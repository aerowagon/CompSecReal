CREATE TABLE sessions (
	id INT auto_increment PRIMARY KEY,
	email varchar(255) NOT NULL,
	sess_token int NOT NULL
);

CREATE TABLE users(
    sys_ID int PRIMARY KEY AUTO_INCREMENT NOT NULL,
    email varchar(255) NOT NULL
);

CREATE TABLE students (
	sys_ID int,
	blugold_ID int NOT NULL,
	email varchar(255) NOT NULL,
	f_name varchar(255) NOT NULL,
	l_name varchar(255) NOT NULL,
	GPA int,
	address varchar(255),
	city varchar(100),
	add_city varchar(100),
	zip_code int,
	PRIMARY KEY (email),
	FOREIGN KEY(sys_ID) REFERENCES users(sys_ID)
);

CREATE TABLE professors (
	sys_ID int,
	email varchar(255) NOT NULL,
	f_name varchar(255) NOT NULL,
	l_name varchar(255) NOT NULL,
	GPA int,
	address varchar(255),
	city varchar(100),
	add_city varchar(100),
	zip_code int,
	PRIMARY KEY(email),
	FOREIGN KEY(sys_ID) REFERENCES users(sys_ID)
);

CREATE TABLE bank (
	sys_ID int NOT NULL,
	credit int,
	FOREIGN KEY (sys_ID) REFERENCES users(sys_ID)
);
