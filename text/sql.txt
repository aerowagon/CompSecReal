

-- first database

DROP DATABASE IF EXISTS compsec_project;
CREATE DATABASE compsec_project;
USE compsec_project;

	CREATE TABLE users(
		sys_ID INT PRIMARY KEY AUTO_INCREMENT NOT NULL,
		email VARCHAR(255) NOT NULL
	);

	CREATE TABLE sessions (
		sess_token varchar(255) PRIMARY KEY NOT NULL,
		permissions INT NOT NULL,
		created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
		last_seen TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
		email varchar(255) NOT NULL,
        
		valid bit NOT NULL
	);

	CREATE TABLE students (
		sys_ID INT, 
		blugold_ID INT AUTO_INCREMENT PRIMARY KEY,
		email VARCHAR(255) NOT NULL,
		f_name VARCHAR(255) NOT NULL,
		l_name VARCHAR(255) NOT NULL,
		GPA FLOAT(3,2) NOT NULL DEFAULT 0.00,
		CONSTRAINT chk_gpa_range CHECK (GPA >= 0.00 AND GPA <= 5.00),
		address VARCHAR(255) NOT NULL,
		city VARCHAR(100) NOT NULL,
		add_city VARCHAR(100) NOT NULL,
		zip_code CHAR(5) NOT NULL,
		CONSTRAINT chk_students_zip_code CHECK (zip_code REGEXP '^[0-9]{5}$'),
		total_credits INT DEFAULT 0,
		FOREIGN KEY (sys_ID) REFERENCES users(sys_ID)

	);

	CREATE TABLE professors (
		sys_ID INT,
		email VARCHAR(255) NOT NULL UNIQUE,
		f_name VARCHAR(255) NOT NULL,
		l_name VARCHAR(255) NOT NULL,
		GPA DOUBLE NOT NULL,
		address VARCHAR(255) NOT NULL,
		city VARCHAR(100) NOT NULL,
		add_city VARCHAR(100) NOT NULL,
		zip_code CHAR(5) NOT NULL,
		CONSTRAINT chk_professors_zip_code CHECK (zip_code REGEXP '^[0-9]{5}$'),
		PRIMARY KEY (email),
		FOREIGN KEY (sys_ID) REFERENCES users(sys_ID)
		
	);

	CREATE TABLE bank (
		sys_ID INT NOT NULL,
		account_balance FLOAT(8,2) NOT NULL DEFAULT 0.00,
		FOREIGN KEY (sys_ID) REFERENCES users(sys_ID)
	);

	CREATE TABLE updates (
		sys_ID INT AUTO_INCREMENT PRIMARY KEY,
		date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
		sess_token varchar(255) NOT NULL,
		updated_table varchar(255) NOT NULL,
		sys_ID_edited INT NOT NULL
	);
    
    CREATE TABLE login_log (
    id INT AUTO_INCREMENT PRIMARY KEY,
    sys_ID INT NOT NULL,
    email VARCHAR(255) NOT NULL,
    session_token VARCHAR(255) NOT NULL,
    login_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    logout_time TIMESTAMP NULL,
    logout_reason ENUM('button','timeout','browser_closed','power_loss') NULL,
    FOREIGN KEY (sys_ID) REFERENCES users(sys_ID)
    );

-- Create an audit table
CREATE TABLE IF NOT EXISTS audit_log (
    id INT AUTO_INCREMENT PRIMARY KEY,
    sys_ID INT,
    table_name VARCHAR(255),
    action_type ENUM('INSERT','UPDATE','DELETE'),
    changed_by_email VARCHAR(255),
    changed_by_token VARCHAR(255),
    column_name VARCHAR(255),
    old_value TEXT,
    new_value TEXT,
    changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	FOREIGN KEY (sys_ID) REFERENCES users(sys_ID)
);

-- Trigger for students table
DELIMITER //
CREATE TRIGGER students_update
AFTER UPDATE ON students
FOR EACH ROW
BEGIN

    IF OLD.email <> NEW.email THEN
        INSERT INTO audit_log 
        (table_name, action_type, changed_by_email, changed_by_token, sys_ID, column_name, old_value, new_value)

DELIMITER //
CREATE TRIGGER students_insert
AFTER INSERT ON students
FOR EACH ROW
BEGIN

DECLARE admin_email VARCHAR(255);

    -- Look up the email from the sessions table using the token
    SELECT email 
    INTO admin_email
    FROM sessions
    WHERE sess_token = @session_user
      AND valid = b'1'   -- optional: only if you want to enforce validity
    LIMIT 1;
    
INSERT INTO audit_log 
(table_name, action_type, changed_by_email, changed_by_token, sys_ID, column_name, old_value, new_value)
VALUES 
('students', 'INSERT', admin_email, @session_user, NEW.sys_ID, 'new row inserted', NULL, NEW.email);
END //
